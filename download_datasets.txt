#!/usr/bin/env python3
"""
Dataset download script for ILS-HED.
Downloads and prepares all datasets used in the paper:
- BSDS500, DRIVE, STARE, DeepCrack, Stone331, SDNET2018
- NYU Depth v2, PASCAL Context (for transfer tests)
"""

import os
import sys
import argparse
import urllib.request
import zipfile
import tarfile
import shutil
import subprocess
from pathlib import Path
from typing import Dict, List, Tuple
import requests
from tqdm import tqdm


class DownloadProgress:
    """Progress bar for downloads."""
    
    def __init__(self, total_size: int, description: str):
        self.pbar = tqdm(total=total_size, unit='B', unit_scale=True, desc=description)
    
    def update(self, downloaded: int, total: int):
        self.pbar.update(downloaded - self.pbar.n)
    
    def close(self):
        self.pbar.close()


def download_file(url: str, destination: Path, description: str = "Downloading"):
    """Download a file with progress bar."""
    response = requests.get(url, stream=True)
    total_size = int(response.headers.get('content-length', 0))
    
    destination.parent.mkdir(parents=True, exist_ok=True)
    
    progress = DownloadProgress(total_size, description)
    with open(destination, 'wb') as f:
        for chunk in response.iter_content(chunk_size=8192):
            f.write(chunk)
            progress.update(progress.pbar.n + len(chunk), total_size)
    progress.close()


def download_bsds500(data_dir: Path):
    """Download BSDS500 dataset."""
    print("\n📥 Downloading BSDS500...")
    
    # BSDS500 from UC Berkeley
    url = "https://www2.eecs.berkeley.edu/Research/Projects/CS/vision/grouping/BSR/BSR_bsds500.tgz"
    output_path = data_dir / "BSR_bsds500.tgz"
    
    download_file(url, output_path, "BSDS500")
    
    # Extract
    print("📦 Extracting...")
    with tarfile.open(output_path, 'r:gz') as tar:
        tar.extractall(data_dir)
    
    # Rename and reorganize
    bsr_dir = data_dir / "BSR"
    bsds_dir = data_dir / "BSDS500"
    if bsr_dir.exists():
        shutil.move(bsr_dir / "BSDS500", bsds_dir)
        shutil.rmtree(bsr_dir)
    
    # Clean up
    output_path.unlink()
    print("✅ BSDS500 downloaded and extracted")

    """Download DRIVE dataset."""
    print("\n📥 Downloading DRIVE...")
    
    # DRIVE from official source
    url = "https://www.isi.uu.nl/Research/Databases/DRIVE/download.php"
    # Note: DRIVE requires manual registration
    # This is a placeholder - actual download requires authentication
    
    print("⚠️  DRIVE requires manual download from https://www.isi.uu.nl/Research/Databases/DRIVE/")
    print("   Please download and extract to:", data_dir / "DRIVE")
    
    # Create directory structure
    drive_dir = data_dir / "DRIVE"

    (drive_dir / "training/images").mkdir(parents=True, exist_ok=True)

    (drive_dir / "test/images").mkdir(parents=True, exist_ok=True)
    (drive_dir / "test/1st_manual").mkdir(parents=True, exist_ok=True)

def download_deepcrack(data_dir: Path):
    """Download DeepCrack dataset."""
    print("\n📥 Downloading DeepCrack...")
    
    # DeepCrack from GitHub
    url = "https://github.com/yhlleo/DeepCrack/releases/download/v1.0/DeepCrack_dataset.zip"
    output_path = data_dir / "DeepCrack.zip"
    
    download_file(url, output_path, "DeepCrack")

        shutil.move(extracted_dir, data_dir / "DeepCrack")

    print("\n📥 Downloading PASCAL Context...")
    
    # PASCAL Context from official source

def verify_dataset(data_dir: Path, dataset_name: str) -> bool:
    """Verify that a dataset exists and has the expected structure."""
    dataset_path = data_dir / dataset_name
    
    if not dataset_path.exists():
        print(f"❌ {dataset_name} not found at {dataset_path}")
        return False
    
    # Basic structure checks
    if dataset_name == "BSDS500":
        required = ["images/train", "images/val", "images/test", 
                    "edges/train", "edges/val", "edges/test"]
    elif dataset_name == "DRIVE":
        required = ["training/images", "training/1st_manual", 
                    "test/images", "test/1st_manual"]
    elif dataset_name == "DeepCrack":
        required = ["train/images", "train/masks", "test/images", "test/masks"]
    else:
        return True
    
    for req in required:
        if not (dataset_path / req).exists():
            print(f"⚠️  {dataset_name} missing: {req}")
            return False
    
    print(f"✅ {dataset_name} verified")
    return True


def main():
    parser = argparse.ArgumentParser(description='Download ILS-HED datasets')
    parser.add_argument('--data_dir', type=str, default='./data', 
                        help='Root directory for datasets')
    parser.add_argument('--datasets', nargs='+', 
                        default=['BSDS500', 'DeepCrack', 'SDNET2018'],
                        choices=['BSDS500', 'DRIVE', 'STARE', 'DeepCrack', 
                                'Stone331', 'SDNET2018', 'NYU_Depth', 'PASCAL_Context'],
                        help='Datasets to download')
    args = parser.parse_args()
    
    data_dir = Path(args.data_dir)
    data_dir.mkdir(parents=True, exist_ok=True)
    
    print("=" * 60)
    print("ILS-HED Dataset Downloader")
    print("=" * 60)
    
    downloaders = {
        'BSDS500': download_bsds500,
        'DRIVE': download_drive,
        'STARE': download_stare,
        'DeepCrack': download_deepcrack,
        'SDNET2018': download_sdnet2018,
        'NYU_Depth': download_nyu_depth,
        'PASCAL_Context': download_pascal_context,
    }
    
    for dataset in args.datasets:
        if dataset in downloaders:
            downloaders[dataset](data_dir)
            verify_dataset(data_dir, dataset)
        else:
            print(f"⚠️  No downloader for {dataset}")
    
    print("\n" + "=" * 60)
    print("Dataset download complete!")
    print("=" * 60)
    
    # Summary
    print("\n📊 Dataset Summary:")
    print(f"  Data directory: {data_dir.absolute()}")
    print("\nNext steps:")
    print("  1. Verify all datasets are correctly structured")
    print("  2. Run ILS search: python -m ils_hed.train --dataset BSDS500 --mode search")
    print("  3. Evaluate: python -m ils_hed.train --dataset BSDS500 --mode eval")


if __name__ == '__main__':
    main()
